meu-projeto-backend/
├── src/
│   ├── config/             # Configurações de ferramentas (Prisma, Mongoose, etc.)
│   ├── controllers/        # Controladores (recebem as requisições HTTP e enviam respostas)
│   ├── database/           # Conexões e scripts de banco de dados (Ex: prisma client)
│   ├── middlewares/        # Middlewares (autenticação, validações, erro global)
│   ├── models/             # Entidades de domínio / Esquemas (Mongoose)
│   ├── repositories/       # Abstração do banco de dados (Ex: findById, create)
│   ├── routes/             # Definição das rotas HTTP
│   ├── services/           # Regras de negócio (onde a mágica acontece)
│   └── server.ts           # Inicialização do servidor Express
├── .env                    # Variáveis de ambiente
├── .gitignore              # Arquivos ignorados pelo Git
├── package.json
└── tsconfig.json
